    Command reference

    Start core layer — Caddy · Postgres · Redis · n8n 
    # Clone and generate secrets (first time only)
git clone https://github.com/SahibYar/open-agency-os.git
cd open-agency-os
bash setup.sh
# Set DOMAIN= and CADDY_EMAIL= in .env, then:
docker compose --profile core up -d
    Add CRM layer — Twenty + Postgres + Redis (already in core) 
    docker compose --profile crm up -d
    Add inbox layer — Chatwoot 
    docker compose --profile inbox up -d
    Backup Postgres — pg_dump to timestamped file 
    docker exec aperture-postgres pg_dumpall -U postgres \
  > backup-$(date +%Y%m%d-%H%M%S).sql
    Upgrade all services — pull latest images and restart 
    # Pull updated images
docker compose pull
# Restart all running profiles
docker compose up -d --remove-orphans
